
local HammerBlaster = {
    lastSpeedRecord = 0,
    
    activeColor = UE.Color(1.5, 1.56, 1.6, 1),
    disactiveColor = UE.Color(0.01, 0.01, 0.01, 1),
    
    hitPuList = {},
    
    hitInterval = 0.5,
    
    isTips = false,
}


function HammerBlaster:Start()
    SetMaterialEmission(self.renderer, self.disactiveColor)
    
    self.fireCore.gameObject:SetActive(false)
    
    self.isOnline = CL.Network.VRNetworkManager.IsOnline
end

function HammerBlaster:OpenSkill()
    self.onSkill = true
    
    self.isTips = false
    
    self:ActiveEffectOn()
end

function HammerBlaster:UpdateSkill()
    if self.isTips then
        return
    end
    
    local ph = self.host.interact:GetPrimaryHand()
    if IsNullOrDestroyed(ph) then
        return
    end

    local character = ph.character
    local needMana = self.skillManaCost
    if character.IsLocalPlayer and character:HasEnoughMana(needMana) == false then
        self.isTips = true
        CL.TextPanel.ShowTipsByTextIdAndCheckInterval("UI_Tips_NoMana", 3)
        return
    end
end

function HammerBlaster:CloseSkill()
    self.onSkill = false
    
    self:ActiveEffectOff()
end

function HammerBlaster:ActiveEffectOn()
    self.activeEffect:Play()
    
    RemoveSchedule(self.host.interact.gameObject, self.emissionTimer)
    self.emissionTimer = SetMaterialEmissionWithinTime(self.host.interact.gameObject, {self.renderer}, self.activeColor, 0.2)
    
    self.fireCore.gameObject:SetActive(true)
    
    self.activeSound:PlaySound()
    self.soundActId = CL.Scheduler.RemoveSchedule(self.soundActId)
    CL.AudioMgr.Instance:GiveBackAndLerpToZero(self.loopSoundInfo)
    
    self.loopSoundInfo = self.loopSound:PlaySoundInfo()
    if self.loopSoundInfo then
        self.loopSoundInfo.ao.transform:SetParent(self.host.interact.trans)
    end
end

function HammerBlaster:ActiveEffectOff()
    self.activeEffect:Stop()
    
    self.fireCore.gameObject:SetActive(false)
    
    if self.loopSoundInfo then
        local volume = self.loopSoundInfo.audio.volume 
        self.soundActId = CL.Scheduler.RemoveSchedule(self.soundActId)
        self.soundActId = CL.Scheduler.Create(self.host.interact, 
            function(sche, t, s)
                if not self.loopSoundInfo then
                    sche:Stop()
                end
                self.loopSoundInfo.audio.volume = self.loopSoundInfo.audio.volume * t/s
                if t >= s then
                    CL.AudioMgr.Instance:GiveBackAndLerpToZero(self.loopSoundInfo)
                    sche:Stop()
                end
            end
        ,0, 0.5).actionId
    end
    
    RemoveSchedule(self.host.interact.gameObject, self.emissionTimer)
    self.emissionTimer = SetMaterialEmissionWithinTime(self.host.interact.gameObject, {self.renderer}, self.disactiveColor, 0.5)

end

function HammerBlaster:TriggerStay(collider)
    if not self.onSkill then
        return
    end
    
    local pu = CL.PhysicsUnit.GetPhysicsUnitTopDown(collider)
    if IsNullOrDestroyed(pu) then
        return
    end
    
    if self.hitPuList[pu] and self.hitPuList[pu] > UE.Time.unscaledTime then
        return
    end
    
    if pu.unitType == CL.Tagger.Tag.InteractRole then
        self.hitPuList[pu] = UE.Time.unscaledTime + self.hitInterval
        self:FireHitScan(collider)
    elseif pu.unitType == CL.Tagger.Tag.PlayerRole then
        self.hitPuList[pu] = UE.Time.unscaledTime + self.hitInterval
        self:FireHitScan(collider)
    end
end

function HammerBlaster:FireHitScan(col)
    local ph = self.host.interact:GetPrimaryHand()
    if IsNullOrDestroyed(ph) then
        return
    end
    
    CL.FullBodyMuscleState.StayDmg = true
    CL.InteractTrigger.BeginScanDmg()
    CL.InteractTrigger.ApplyHitScanDamage(col, self.host.interact, self.host.interact.rb, self.fireCore, self.host.interact.trans.forward, ph.character.aiProxy:GetCamp())
    CL.InteractTrigger.EndHitScan()
    CL.FullBodyMuscleState.StayDmg = false
end

----------------------------------------------------------------------------------------

function HammerBlaster:OnColliderEnter(collision)
    if self.isOnline then
        if not SafeLocalTrackPhysics(self.host.interact) then
            return
        end
    end
    
    if not self.onSkill then
        return
    end
    
    if collision.collider.gameObject.layer ~= CL.LayerDefine.EnvLayer.value then
        return
    end

    -- cost mana
    local ph = self.host.interact:GetPrimaryHand()
    if IsNullOrDestroyed(ph) then
        return
    end
    
    --check skill condition
    local character = ph.character
    local needMana = self.skillManaCost * self.host.AppliedManaCostMultiplier
    if character:HasEnoughMana(needMana) == false then
        return
    end
    
    local relaVel = collision.relativeVelocity
    local relaVelY = UE.Mathf.Abs(relaVel.y)
    if relaVelY < self.impactSpeed then
        return
    end

    local thisCol, otherCol, point, normal = collision:GetContactPoint(0)

    local dot = UE.Vector3.Dot(normal, UE.Vector3.up)
    if dot < 0.5 then
        return
    end
    
    --do releaseSkill
    if self.isOnline then
        if CL.Network.VRNetworkManager.ServerOnline then
            self:DoReleaseSkill(ph.character, needMana, point)
        else
            local state = GetNetworkIBState(self)
            if state ~= nil then state:CommandReleaseSkill(ph.character.networkPlayer.netId, point) end
        end
    else
        self:DoReleaseSkill(ph.character, needMana, point)
    end
    
    self.host:CloseSkill()
end

function HammerBlaster:NetworkReleaseSkill(character, point)
    local needMana = self.skillManaCost * self.host.AppliedManaCostMultiplier
    if character:HasEnoughMana(needMana) == false then
        return
    end
    
    character:CostMana(self.host, needMana)
    -- cost mana
    
    self:CreateGroundWave(point)
    self.host:CloseSkill()
end

function HammerBlaster:DoReleaseSkill(character, mana, point)
    character:CostMana(self.host, mana)
    -- cost mana
    
    self:CreateGroundWave(point)
    if self.isOnline then
        local state = GetNetworkIBState(self)
        if state ~= nil then state:ServerCloseSkill() end
    end
end


function HammerBlaster:GetGroundPoint(position)
    local isHit, raycastHit  = UE.Physics.SphereCast(position + UE.Vector3.up * 0.5, 0.05, UE.Vector3.down, 1, CL.LayerDefine.EnvLayerMask.value)
    --if isHit and UE.Vector3.Angle(raycastHit.normal, UE.Vector3.up) < 10 then
    if isHit then
        return raycastHit.point
    else
        return nil
    end
end

function HammerBlaster:CreateGroundWave(point)
    local pos = self:GetGroundPoint(point)
    if not pos then
        return
    end
    
    local dir = UE.Vector3(self.host.interact.trans.forward.x, 0, self.host.interact.trans.forward.z)
    dir = dir.normalized
    
    self.strikeSound:PlaySoundInfo()

    local list = {}
    local idx = 1
    while self["flyObj" .. idx] do
        table.insert(list, self["flyObj" .. idx])
        idx = idx + 1
    end
    
    if #list <= 0 then
        return
    end
    
    if self.host.interact:IsEnhance() then
        idx = 1
        while self["flyObj_enhance" .. idx] do
            table.insert(list, self["flyObj_enhance" .. idx])
            idx = idx + 1
        end
    end
    
    local fo = nil
    if self.isOnline then
        if CL.Network.VRNetworkManager.Instance.IsServer then
            fo = self.host:NetworkCreateFlyObj(list[1], pos + UE.Vector3.up * 0.005, dir)
        else
            return
        end
    else
        fo = self.host:CreateFlyObj(list[1], pos + UE.Vector3.up * 0.005, dir)
    end
    table.remove(list, 1)
    
    if #list > 0 then
        if fo then
            fo.script.self:SetInfo(dir, self.interval, list)
        end
    end
end



return Class(nil, nil, HammerBlaster)
