
local FlammableTorch = {
    curIntensity = 0,
    
    burningSpeed = 2,
    
    hitScanInterval = 1,
    hitEndTime = 0,
    
    activeColor = UE.Color(2.43, 0.86, 0, 1),
    disactiveColor = UE.Color(0.54, 0.19, 0.1, 1),
    
    hitInfos = {},
}

function FlammableTorch:Start()
    SetMaterialEmission(self.renderer, self.disactiveColor)

    if not IsNullOrDestroyed(self.light) then
        self.light.intensity = self:GetDefaultLightIntensity()
    end
end

function FlammableTorch:OnGrab(a, g)
    if g then
        self.hand = a.attachedHand
    end
end

function FlammableTorch:OpenSkill()
    if CL.Network.VRNetworkManager.IsOnline then
        if CL.Network.VRNetworkManager.ServerOnline then
            self.delayToCloseId = CL.Scheduler.RemoveSchedule(self.delayToCloseId)
            local state = GetNetworkIBState(self)
            if state ~= nil then state:ServerStartBurning() end
        end
        return
    end
    self:StartBurning()
end

function FlammableTorch:CloseSkill()
    if CL.Network.VRNetworkManager.IsOnline and not CL.Network.VRNetworkManager.ServerOnline then
        return
    end
    self.delayToCloseId = CL.Scheduler.RemoveSchedule(self.delayToCloseId)
    self.delayToCloseId = CL.Scheduler.Create(self.trans,
        function(sche, t, s)
            if IsNullOrDestroyed(self.host) then return end
            if CL.Network.VRNetworkManager.IsOnline then
                local state = GetNetworkIBState(self)
                if state ~= nil then state:ServerStopBurning() end
            else
                self:StopBurning()
            end
        end
    , 0, 0, self.delayTime).actionId
end

function FlammableTorch:StartBurning()
    self.isBurning = true
    
    self.light.range = self.burningLightRange
    self.light.intensity = self.burningLightIntensity
    
    CL.Scheduler.RemoveSchedule(self.delayToCloseId)
    
    local soundInfo = self.lunchSound:PlaySoundInfo()
    if soundInfo then
        soundInfo.ao.transform:SetParent(self.host.interact.trans)
    end
    --effect
    self.flameEffect:Play()
    local color = nil
    local lastColor = self.renderer.material:GetColor("_EmissionColor")
    
    self.burningActId = CL.Scheduler.RemoveSchedule(self.burningActId)
    self.burningActId = CL.Scheduler.Create(self.host.interact,
        function(sche, t, s)
            if IsNullOrDestroyed(self.host) then sche:Stop(); return end
            if self.curIntensity < 1 then
                self.curIntensity = math.min(self.curIntensity 
                    + UE.Time.unscaledDeltaTime * self.burningSpeed, 1)
                
            
                --update effect
                color = UE.Color.Lerp(lastColor, self.activeColor, self.curIntensity)
                SetMaterialEmission(self.renderer, color)
            end
        end
    , 0, -1)
    :SetUpdateChannel(CL.Scheduler.UpdateChannel.FixedUpdate).actionId
end

function FlammableTorch:StopBurning()
    self.isBurning = true
    
    local endTime = self.curIntensity/self.burningSpeed
    if not IsNullOrDestroyed(self.flameEffect) then
        self.flameEffect:Stop()
    end
    
    if IsNullOrDestroyed(self.renderer) then
        return
    end
    
    local color = nil
    local lastColor = self.renderer.material:GetColor("_EmissionColor")
    self.burningActId = CL.Scheduler.RemoveSchedule(self.burningActId)
    self.burningActId = CL.Scheduler.Create(self.host.interact,
        function(sche, t, s)
            if IsNullOrDestroyed(self.renderer) or IsNullOrDestroyed(self.light) then
                return
            end
    
            self.curIntensity = self.curIntensity - UE.Time.unscaledDeltaTime * self.burningSpeed
            if self.curIntensity > 0 then
                --update effect
                color = UE.Color.Lerp(lastColor, self.disactiveColor, 1 - self.curIntensity)
                SetMaterialEmission(self.renderer, color)
                
                self.light.range = self.defaultLightRange + self.curIntensity * 
                    (self.burningLightRange - self.defaultLightRange)
                self.light.intensity = self:GetDefaultLightIntensity() + self.curIntensity * 
                    (self.burningLightIntensity - self:GetDefaultLightIntensity())
            end
            
            if t >= endTime or self.curIntensity <= 0 then
                self.isBurning = false
                
                self.light.range = self.defaultLightRange
                self.light.intensity = self:GetDefaultLightIntensity()
                
                SetMaterialEmission(self.renderer, self.disactiveColor)
                
                sche:Stop()
            end
        end
    , 0, endTime)
    :SetUpdateChannel(CL.Scheduler.UpdateChannel.FixedUpdate).actionId
end

function FlammableTorch:OnColliderStay(collision)
    if not self.isBurning then
        return
    end
    local pu = CL.PhysicsUnit.GetPhysicsUnitTopDown(collision.transform)
    if pu then
        local unitType = pu.unitType
        if unitType == CL.Tagger.Tag.InteractRole then
            if pu.attr:IsImmune("Fire") then
                return
            end
            
            if not pu.muscleState:CanIgnite(collision.collider) then
                return
            end
        end
        
        self:TryIgnite(pu, collision.collider)

        if self.hitInfos[pu] and self.hitInfos[pu] > UE.Time.time then
            return
        end
        
        if unitType == CL.Tagger.Tag.SceneObject
            or unitType == CL.Tagger.Tag.InteractRole
            or unitType == CL.Tagger.Tag.InteractObj
            or unitType == CL.Tagger.Tag.PlayerRole
            or unitType == CL.Tagger.Tag.PlayerHand
            or unitType == CL.Tagger.Tag.InteractWeapon then
            
            local setupToSelf = false
            if unitType == CL.Tagger.Tag.PlayerRole then
                if self.hand and self.hand.character == pu.character then
                    -- return
                    setupToSelf = true
                end
            end
            local thisCol, otherCol, point, normal = collision:GetContactPoint(0)
            --self:BurningTarget(point, normal)
            self.hitInfos[pu] = UE.Time.time + self.hitScanInterval
            
            self:SetupFlame(collision.transform, collision.collider:ClosestPoint(point), setupToSelf, collision.collider)
            
            if unitType == CL.Tagger.Tag.InteractRole then
                pu.muscleState:RecordIgnite(collision.collider)
            end
        end
    end
end

function FlammableTorch:TryIgnite(pu, col)
    if pu.unitType ~= CL.Tagger.Tag.InteractObj and pu.unitType ~= CL.Tagger.Tag.InteractWeapon then
        return
    end
        
    local trigger = pu:GetInteractTrigger()
    if IsNullOrDestroyed(trigger) or not trigger.script then
        return
    end
        
    local tb = trigger.script:GetSelf()
    if tb.fuseCol and tb.Ignite then
        tb:Ignite()
    end
end

function FlammableTorch:BurningTarget(point, dir)
    self.host:CreateFlyObj(self.flyObjName, point, -dir, 0)
    CL.EffectMgr.Instance:PlayEffect(self.hitFireEffect, point, UE.Quaternion.LookRotation(dir))
end

function FlammableTorch:SetupFlame(trans, pos, setupToSelf, targetCol)
    local isOnline = CL.Network.VRNetworkManager.IsOnline
    if isOnline and not CL.Network.VRNetworkManager.ServerOnline then return nil end

    local fo = nil
    if isOnline then
        if setupToSelf then
            fo = CL.Network.NetworkSpawnMgr.Instance:NetworkCreateFlyObj(self.zoneObjName, nil, pos, UE.Vector3.up)
        else
            fo = self.host:NetworkCreateFlyObj(self.zoneObjName, pos, UE.Vector3.up)
        end
        trans = targetCol.transform
    elseif not setupToSelf then
        fo = self.host:CreateFlyObj(self.zoneObjName, pos, UE.Vector3.up)
    else
        fo = CL.FlyObject.CreateFlyObj(self.zoneObjName, nil, pos, UE.Vector3.up)
    end

    if IsNullOrDestroyed(fo) then return nil end
    local follower = fo.trans:GetComponent(typeof(CL.FlyObjectFollower))
    if not IsNullOrDestroyed(follower) then
        local localPos = trans:InverseTransformPoint(pos)
        follower:SetData(trans, localPos)
        if isOnline and not IsNullOrDestroyed(fo.networkFO) then
            local colKey = CL.Network.NetworkColliderMgr.Instance:GetColliderKey(targetCol)
            if colKey >= 0 then
                fo.networkFO:ServerInitZoneByCol(targetCol, localPos)
            else
                fo.networkFO:ServerInitZone(pos)
            end
        end
    end
    
    return fo
end

function FlammableTorch:GetDefaultLightIntensity()
    if not IsNullOrDestroyed(CL.MixedRealityMgr.Instance) then
        return 0
    else
        return self.defaultLightIntensity
    end
end

return Class(nil, nil, FlammableTorch)
