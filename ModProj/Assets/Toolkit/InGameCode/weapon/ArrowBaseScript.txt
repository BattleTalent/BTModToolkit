--luaBehaviour
local ArrowBase = {
    lifeTime = 10,
    }

function ArrowBase:Awake()
    self.onStabEnterHandler = function(si)
        CL.Scheduler.RemoveSchedule(self.delayTrailId)
        if not IsNullOrDestroyed(self.trail) then
            self.trail.enabled = false
        end
    end
    
    self.isOnline = not IsNullOrDestroyed(CL.Network.VRNetworkManager.Instance) 
        and not CL.Network.VRNetworkManager.Instance.IsOffline
end

function ArrowBase:Reset()
    self.trail = self.host.gameObject:GetComponent(UE.TrailRenderer)
end

function ArrowBase:StartLifeCircle()
    self:StopLifeCircle()
    
    if self.isOnline then
        if CL.Network.NetworkRoomMgr.IsInPvpMode then
            if not SafeLocalTrackPhysics(self.ib) then
                return
            end
        else
            if not CL.Network.VRNetworkManager.Instance.IsServer then
                return
            end
        end
    end
    
    CL.Scheduler.RemoveSchedule(self.lifeCircleId)    
    self.lifeCircleId = CL.Scheduler.Create(self.ib, 
        function(sche, t, s)
            self:Update()
        end
    , 0, -1)
    :SetUpdateChannel(CL.Scheduler.UpdateChannel.FixedUpdate).actionId
end

function ArrowBase:StopLifeCircle()
    CL.Scheduler.RemoveSchedule(self.lifeCircleId)
    self.lifeCircleId = nil
end


function ArrowBase:OnEnable()
    self:StartLifeCircle()
    
    
    local so = self.ib:GetStabObject()
    if not IsNullOrDestroyed(so) then
        so:StabEnterEvent("+", self.onStabEnterHandler)
    end
end

function ArrowBase:OnDisable()
    self:StopLifeCircle()
    
    local so = self.ib:GetStabObject()
    if not IsNullOrDestroyed(so) then
        so:StabEnterEvent("-", self.onStabEnterHandler)
    end
end


function ArrowBase:Update()
    self:AdjectRotation()
end

function ArrowBase:AdjectRotation()
    if IsNullOrDestroyed(self.ib) then
        return
    end
    
    if self.isOnline then
        if not SafeLocalTrackPhysics(self.ib) then
            return
        end
    end
    
    if not self.disabled then
        if not IsNullOrDestroyed(self.ib.rb) and self.ib.rb.velocity.sqrMagnitude > 0.01 then
            CL.PhysicsFollow.TrackRotation(self.ib.rb, UE.Quaternion.LookRotation(self.ib.rb.velocity))
        end
    end
end

function ArrowBase:OnPut(bow)
    if not IsNullOrDestroyed(bow.host) and not IsNullOrDestroyed(bow.host.interact) then
        self.ib:SetLatestHolder(bow.host.interact:GetLatestHolder())
    end
end

function ArrowBase:OnShoot(bow)
    if not IsNullOrDestroyed(self.trail) then
        self.trail.enabled = true
    end
    
    self.disabled = false

    if self.isOnline and not CL.Network.VRNetworkManager.Instance.IsServer and GetIBNetwork(self.ib) ~= nil then
        return
    end
    CL.FlyObject.CopyWeaponBuffToUnit(bow.host.interact, self.ib)
    
    CL.Scheduler.RemoveSchedule(self.givebackId)    
    self.givebackId = CL.Scheduler.Create(self.ib, 
        function(sche, t, s)
            self:GiveBack()
        end,
    0, 0, self.lifeTime).actionId
end

function ArrowBase:OnNpcShoot()
    if self.isOnline and CL.Network.VRNetworkManager.ServerOnline then
        local state = GetIBNetworkState(self.ib)
        if state ~= nil then state:ServerOnShoot() end
    end

    if not IsNullOrDestroyed(self.trail) then
        self.trail.enabled = true
    end
    self.disabled = false

    if self.isOnline and not CL.Network.VRNetworkManager.ServerOnline then
        return
    end
    CL.Scheduler.RemoveSchedule(self.givebackId)    
    self.givebackId = CL.Scheduler.Create(self.ib, 
        function(sche, t, s)
            self:GiveBack()
        end,
    0, 0, self.lifeTime).actionId
end

function ArrowBase:DelayToShowTrail()
    CL.Scheduler.RemoveSchedule(self.delayTrailId)
    self.delayTrailId = CL.Scheduler.Create(self.ib, 
        function(sche, t, s)
            if not IsNullOrDestroyed(self.trail) then
                self.trail.enabled = true
            end
        end,
    0, 0, 0.02).actionId
end

function ArrowBase:NetworkOnShoot()
    if self.isOnline and not CL.Network.VRNetworkManager.Instance.IsServer then
        self:DelayToShowTrail()
    else
        if not IsNullOrDestroyed(self.trail) then
            self.trail.enabled = true
        end
    end
    
    self.disabled = false
end

function ArrowBase:OnColliderEnter(collision)
    --if self.isOnline and not CL.Network.VRNetworkManager.Instance.IsServer then
    --    return
    --end
    
    if self.disabled then
        return
    end
    
    local hand = CL.InteractHand.GetHandByCollider(collision.collider)
    if not IsNullOrDestroyed(hand) then
        return
    end
    
    local player = CL.PlayerUnit.GetPhysicsUnit(collision.transform.root)
    if not IsNullOrDestroyed(player) then
        if player.unitType == CL.Tagger.Tag.PlayerRole then
            return
        end
    end
    
    if self.isOnline then
        if SafeLocalTrackPhysics(self.ib) then
            local state = GetIBNetworkState(self.ib)
            if state ~= nil then
                if CL.Network.VRNetworkManager.ServerOnly then
                    state:ServerOnCollision()
                else
                    state:CommandOnCollision()
                end
            end
        end
    end
    
    self:OnCollision()
end

function ArrowBase:OnCollision()
    self:DisableTrail()
    self.disabled = true
end

function ArrowBase:DisableTrail()
    CL.Scheduler.RemoveSchedule(self.delayTrailId)
    if not IsNullOrDestroyed(self.trail) then
        self.trail.enabled = false
    end
end

function ArrowBase:OnCancel(bow)
    self:GiveBack()
end

function ArrowBase:GiveBack()
    if self.isOnline then
        if GetIBNetwork(self.ib) ~= nil then
            if CL.Network.VRNetworkManager.Instance.IsServer then
                local state = GetIBNetworkState(self.ib)
                if state ~= nil then state:ServerGiveBack() end
            end
        end
    end
    self:DoGiveBack()
end

function ArrowBase:DoGiveBack()
    self.disabled = false
    if not IsNullOrDestroyed(self.ib) then
        self.ib:SetRBCols(true)
        self.ib:SetKinematic(false)
        self.ib:Clear()
    end
    
    self:StopLifeCircle()
    CL.Scheduler.RemoveSchedule(self.givebackId)
    CL.Scheduler.RemoveSchedule(self.delayTrailId)
    
    self:OnResetState()
    
    if self.trail ~= nil and not IsNullOrDestroyed(self.trail) then
        self.trail:Clear()
        self.trail.enabled = false
    end
    if not IsNullOrDestroyed(self.ib) then
        local hitInfoObj = self.ib:GetRagdollHitInfoObj()
        if hitInfoObj then
            hitInfoObj:ResetState()
        end
        if self.isOnline then
            if not IsNullOrDestroyed(self.ib) then
                local networkIB = GetIBNetwork(self.ib)
                if networkIB ~= nil then
                    networkIB:OnGiveBack()
                end
            end
        end
        if self.ib:IsOnEnchant() then
            self.ib:RemoveEnchant()
        end
        CL.PrefabPool.GiveBack(self.ib.gameObject)
    end
end

function ArrowBase:ResetArrow()
    self.disabled = false
    
    if not IsNullOrDestroyed(self.ib) then
        self.ib.trans.localScale = UE.Vector3.one
        self.ib:SetRBCols(true)
        self.ib:SetKinematic(false)
        self.ib:Clear()
    end
    
    self:StopLifeCircle()
    CL.Scheduler.RemoveSchedule(self.givebackId)
    CL.Scheduler.RemoveSchedule(self.delayTrailId)
    
    self:OnResetState()
    
    if self.trail ~= nil and not IsNullOrDestroyed(self.trail) then
        self.trail:Clear()
        self.trail.enabled = false
    end
    if not IsNullOrDestroyed(self.ib) then
        local hitInfoObj = self.ib:GetRagdollHitInfoObj()
        if hitInfoObj then
            hitInfoObj:ResetState()
        end
        
        if self.ib:IsOnEnchant() then
            self.ib:RemoveEnchant()
        end
    end
end

function ArrowBase:OnResetState()

end

return Class(nil, nil, ArrowBase)
