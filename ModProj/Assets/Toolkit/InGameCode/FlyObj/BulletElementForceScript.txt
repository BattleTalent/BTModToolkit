function BulletElementForce:GetForwardTarget()
    local isHit, raycastHit  = UE.Physics.SphereCast(self.trans.position + self.trans.forward * 0.1, 0.05, self.trans.forward, self.maxRange,
        UE.LayerMask.GetMask(
                UE.LayerMask.LayerToName(CL.LayerDefine.EnvLayer.value), 
                UE.LayerMask.LayerToName(CL.LayerDefine.RagdollLayer.value), 
                UE.LayerMask.LayerToName(CL.LayerDefine.RagdollFakeLayer.value), 
                UE.LayerMask.LayerToName(CL.LayerDefine.InteractLayer.value),
                UE.LayerMask.LayerToName(CL.LayerDefine.CharacterLayer.value)))
    if not isHit then
        return nil, nil
    end
    
    return raycastHit.collider.transform, raycastHit.point
end

function BulletElementForce:GetPerpendicularPoint(target)
    local ba = target - self.trans.position
    local nbc = self.trans.forward
    local dbp = UE.Vector3.Dot(ba, nbc)
    return self.trans.position + dbp * nbc
end

function BulletElementForce:StartTracking(trans, point)
    local localPoint = trans:InverseTransformPoint(point)
    local targetPoint = trans:TransformPoint(localPoint)
    local targetDir = nil
    local rotIntensity = 0
    local maxIntensity = 0
    local maxDis = UE.Vector3.Distance(self.trans.position, targetPoint)
    local dis = nil
    CL.Scheduler.RemoveSchedule(self.trackingActId)
    self.trackingActId = CL.Scheduler.Create(self.trans, 
        function(sche, t, s)
            if IsNullOrDestroyed(trans) then
                sche:Stop()
                return
            end
            targetPoint = trans:TransformPoint(localPoint)
            targetDir = (targetPoint - self.trans.position).normalized
            dis = UE.Vector3.Distance(self.trans.position, targetPoint)
            if dis <= self.host.rb.velocity.magnitude * UE.Time.deltaTime then
                self.host.trans.position = targetPoint
                sche:Stop()
                return
            end
            rotIntensity = (1 - math.min(dis/maxDis, 1))
            if maxIntensity > rotIntensity then
                rotIntensity = maxIntensity
            else
                maxIntensity = rotIntensity
            end
            
            self.trans.forward = UE.Vector3.Slerp(self.trans.forward, targetDir, math.min( UE.Time.deltaTime * (self.rotAdd + self.rotMlp * rotIntensity), 1))
            
            --print("rot-----------:" ..  "int:" .. rotIntensity .. "  " .. UE.Time.deltaTime * (self.rotAdd + self.rotMlp * rotIntensity))
            self.host.rb.velocity = self.trans.forward * self.host.force
            self.host.rb.angularVelocity = UE.Vector3.zero
        end,
    0, self.host.maxFlyTime)
    :SetUpdateChannel(CL.Scheduler.UpdateChannel.FixedUpdate).actionId
end

    self.isUsed = false
